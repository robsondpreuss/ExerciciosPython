ex02_bugs.py
import pandas as pd
import matplotlib.pyplot as plt
from flask import Flask

app = Flask(__name__)

def grafico():
    df = pd.read_csv("vendas.csv")
    df.groupby("produto")["total"].sum().plot(kind="bar")
    plt.show()
    plt.savefig("grafico.png")

@app.route("/")
def painel():
    return f"<style>body { background: red }</style><h1>Painel</h1>"

def api():
    return jsonify({"ok": True})

app.run(debug=True)
🔑 Ver solução

Só abra depois de tentar!

Os 5 bugs: (1) plt.show() num app web abre janela no SERVIDOR e trava tudo — em dashboard, NUNCA use show; use savefig (e o modo "Agg", que você verá na Etapa 3). (2) savefig DEPOIS do show = PNG em branco (Semana 18!). (3) O gráfico salvou fora da pasta static — o Flask não vai servi-lo. (4) Chaves SIMPLES no CSS dentro da f-string → erro; precisa {{ }}. (5) A função api() está sem o decorador @app.route("/api") (e sem importar jsonify) — sem o decorador, a rota simplesmente não existe.

etapa3.py
import matplotlib
matplotlib.use("Agg")   # modo servidor: gera PNG sem abrir janela
import matplotlib.pyplot as plt

def gerar_graficos(df):
    por_produto = df.groupby("produto")["total"].sum()
    plt.figure(figsize=(6, 4))
    plt.bar(por_produto.index, por_produto.values, color="#3776AB")
    plt.title("Faturamento por produto")
    plt.ylabel("R$")
    plt.tight_layout()
    plt.savefig("static/por_produto.png", dpi=120)
    plt.close()

    por_dia = df.groupby("data")["total"].sum()
    plt.figure(figsize=(6, 4))
    plt.plot(por_dia.index, por_dia.values, marker="o", color="#FFB000")
    plt.title("Faturamento por dia")
    plt.ylabel("R$")
    plt.tight_layout()
    plt.savefig("static/por_dia.png", dpi=120)
    plt.close()
    print("2 gráficos gerados em static/ ✔")

app.py
import os
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from flask import Flask, jsonify, redirect

app = Flask(__name__)

def preparar():
    os.makedirs("dados", exist_ok=True)
    os.makedirs("static", exist_ok=True)

def gerar_exemplo():
    caminho = os.path.join("dados", "vendas.csv")
    if os.path.exists(caminho):
        return
    conteudo = """data,produto,qtd,total
10/06/2026,caneca branca,3,75.0
10/06/2026,caneca preta,12,270.0
11/06/2026,azulejo,2,80.0
11/06/2026,caneca preta,5,137.5
12/06/2026,caneca branca,4,100.0
12/06/2026,azulejo,1,40.0
"""
    with open(caminho, "w", encoding="utf-8") as arquivo:
        arquivo.write(conteudo)

def carregar_dados():
    return pd.read_csv(os.path.join("dados", "vendas.csv"))

def analisar(df):
    return {
        "faturamento": float(df["total"].sum()),
        "unidades": int(df["qtd"].sum()),
        "campeao": df.groupby("produto")["qtd"].sum().idxmax(),
        "ticket": round(float(df["total"].mean()), 2)
    }

def gerar_graficos(df):
    por_produto = df.groupby("produto")["total"].sum()
    plt.figure(figsize=(6, 4))
    plt.bar(por_produto.index, por_produto.values, color="#3776AB")
    plt.title("Faturamento por produto")
    plt.ylabel("R$")
    plt.tight_layout()
    plt.savefig("static/por_produto.png", dpi=120)
    plt.close()

    por_dia = df.groupby("data")["total"].sum()
    plt.figure(figsize=(6, 4))
    plt.plot(por_dia.index, por_dia.values, marker="o", color="#FFB000")
    plt.title("Faturamento por dia")
    plt.ylabel("R$")
    plt.tight_layout()
    plt.savefig("static/por_dia.png", dpi=120)
    plt.close()

@app.route("/")
def painel():
    r = analisar(carregar_dados())
    return f"""
    <html><head><title>Dashboard Tua Caneca</title>
    <style>
      body {{ font-family: Arial; background: #F5F8FC; margin: 40px; }}
      h1 {{ color: #3776AB; }}
      .cartao {{ background: white; display: inline-block; padding: 18px 28px;
                 margin: 8px; border-radius: 12px; box-shadow: 0 2px 8px #0002; }}
      img {{ border-radius: 12px; margin: 10px; box-shadow: 0 2px 8px #0002; }}
    </style></head><body>
    <h1>☕ Dashboard Tua Caneca</h1>
    <div class="cartao"><b>Faturamento</b><br>R$ {r["faturamento"]:.2f}</div>
    <div class="cartao"><b>Unidades</b><br>{r["unidades"]}</div>
    <div class="cartao"><b>Campeão</b><br>{r["campeao"]}</div>
    <div class="cartao"><b>Ticket médio</b><br>R$ {r["ticket"]:.2f}</div>
    <br>
    <img src="/static/por_produto.png" width="440">
    <img src="/static/por_dia.png" width="440">
    <br><a href="/atualizar">🔄 Atualizar</a> · <a href="/api/resumo">API JSON</a>
    </body></html>
    """

@app.route("/api/resumo")
def api_resumo():
    return jsonify(analisar(carregar_dados()))

@app.route("/atualizar")
def atualizar():
    gerar_graficos(carregar_dados())
    return redirect("/")
