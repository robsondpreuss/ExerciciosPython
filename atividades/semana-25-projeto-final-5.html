Os 5 bugs: (1) a rota recebe formulário mas não declara methods=["POST"] — só aceitaria GET. (2) "INSERT ... VALUES (" + nome + ")" é o pecado mortal: concatenação = SQL injection! Use ? (Semana 21). (3) faltou conexao.commit() — sem ele o INSERT não é salvo. (4) redirect foi usado mas NÃO importado (from flask import ..., redirect). (5) a conexão nunca é fechada (conexao.close()).

solucao_ex02.py
from flask import Flask, request, redirect
import sqlite3

app = Flask(__name__)

@app.route("/salvar", methods=["POST"])
def salvar():
    nome = request.form.get("nome")
    conexao = sqlite3.connect("loja.db")
    cursor = conexao.cursor()
    cursor.execute("INSERT INTO produtos (nome) VALUES (?)", (nome,))
    conexao.commit()
    conexao.close()
    return redirect("/")

app.run(debug=True)

Esses 5 são os erros mais comuns de quem começa com Flask + banco. Decorá-los economiza horas de depuração!

etapa1.py
import sqlite3

def iniciar_banco():
    conexao = sqlite3.connect("loja.db")
    cursor = conexao.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS produtos (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nome TEXT, preco REAL, estoque INTEGER
        )
    """)
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS vendas (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            produto TEXT, qtd INTEGER, total REAL
        )
    """)
    cursor.execute("SELECT COUNT(*) FROM produtos")
    if cursor.fetchone()[0] == 0:
        cursor.executemany(
            "INSERT INTO produtos (nome, preco, estoque) VALUES (?, ?, ?)",
            [("caneca branca", 25.0, 10),
             ("caneca preta", 27.5, 8),
             ("azulejo", 40.0, 5)])
        print("Banco inicializado ✔ (3 produtos de exemplo)")
    conexao.commit()
    conexao.close()

etapa4.py
from flask import Flask, request
import sqlite3

@app.route("/vender", methods=["GET", "POST"])
def vender():
    if request.method == "POST":
        nome = request.form.get("nome")
        qtd = int(request.form.get("qtd"))

        conexao = sqlite3.connect("loja.db")
        cursor = conexao.cursor()
        cursor.execute("SELECT preco, estoque FROM produtos WHERE nome = ?", (nome,))
        r = cursor.fetchone()

        if r is None:
            msg = "Produto não encontrado!"
        elif r[1] < qtd:
            msg = f"Estoque insuficiente! Disponível: {r[1]}"
        else:
            preco, estoque = r
            total = qtd * preco
            cursor.execute(
                "UPDATE produtos SET estoque = estoque - ? WHERE nome = ?",
                (qtd, nome))
            cursor.execute(
                "INSERT INTO vendas (produto, qtd, total) VALUES (?, ?, ?)",
                (nome, qtd, total))
            conexao.commit()
            msg = f"Venda registrada! Total: R$ {total:.2f}"
        conexao.close()
        return f"<h1>{msg}</h1><a href='/vender'>← Nova venda</a> | <a href='/'>Catálogo</a>"

    return """
    <h1>Registrar venda</h1>
    <form method="post" action="/vender">
        <input name="nome" placeholder="Produto">
        <input name="qtd" placeholder="Quantidade">
        <button>Vender</button>
    </form><a href="/">← Catálogo</a>
    """

etapa5.py
from flask import Flask, jsonify
import sqlite3

@app.route("/relatorio")
def relatorio():
    conexao = sqlite3.connect("loja.db")
    cursor = conexao.cursor()

    cursor.execute("SELECT SUM(total), COUNT(*) FROM vendas")
    faturamento, qtd = cursor.fetchone()
    faturamento = faturamento or 0

    cursor.execute("""
        SELECT produto, SUM(qtd) FROM vendas
        GROUP BY produto ORDER BY SUM(qtd) DESC
    """)
    ranking = cursor.fetchall()
    conexao.close()

    html = f"<h1>📊 Relatório</h1><p>Faturamento: R$ {faturamento:.2f} em {qtd} vendas</p>"
    html += "<h2>Ranking</h2><ul>"
    for produto, unidades in ranking:
        html += f"<li>{produto}: {unidades} un.</li>"
    html += "</ul><a href='/'>← Catálogo</a>"
    return html

@app.route("/api/produtos")
def api_produtos():
    conexao = sqlite3.connect("loja.db")
    cursor = conexao.cursor()
    cursor.execute("SELECT nome, preco, estoque FROM produtos")
    produtos = [{"nome": n, "preco": p, "estoque": e}
                for n, p, e in cursor.fetchall()]
    conexao.close()
    return jsonify(produtos)
