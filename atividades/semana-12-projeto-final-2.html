Caça aos bugs de POO 🐛🔍

⭐⭐⭐ desafio
🎯 Objetivo

Encontrar os 5 bugs clássicos de quem está começando com classes.

🧭 Passo a passo
  1. O código deveria criar um produto e aplicar 10% de desconto — mas tem 5 bugs
  2. Encontre todos antes de rodar, depois corrija um por um
ex02_bugs.py
class Produto:
    def _init_(self, nome, preco):
        nome = nome
        self.preco = preco

    def aplicar_desconto(percentual):
        preco = preco * (1 - percentual)

caneca = Produto("Caneca", 25)
print(caneca.nome)
caneca.aplicar_desconto(0.1)
print(caneca.preco)
🔑 Ver solução

Só abra depois de tentar!

Os 5 bugs: (1) _init_ com UM underline não é o __init__ — vira método comum e o Python reclama que Produto() não aceita argumentos! (2) nome = nome sem self. não cria atributo — o nome se perde. (3) aplicar_desconto(percentual) esqueceu o self. (4) Dentro dele, preco sem self. duas vezes — não altera o objeto. (5) Consequência: caneca.nome daria AttributeError mesmo se chegasse lá.

solucao_ex02.py
class Produto:
    def __init__(self, nome, preco):
        self.nome = nome
        self.preco = preco

    def aplicar_desconto(self, percentual):
        self.preco = self.preco * (1 - percentual)

caneca = Produto("Caneca", 25)
print(caneca.nome)
caneca.aplicar_desconto(0.1)
print(caneca.preco)

Saída correta: Caneca / 22.5. Memorize a dupla fatal: __init__ com dois underlines e self em tudo.

etapa1.py
class Produto:
    def __init__(self, nome, preco, estoque):
        self.nome = nome
        self.preco = preco
        self.estoque = estoque

    def vender(self, qtd):
        if qtd <= 0 or qtd > self.estoque:
            return False
        self.estoque = self.estoque - qtd
        return True

    def to_dict(self):
        return {"nome": self.nome, "preco": self.preco, "estoque": self.estoque}

    def __str__(self):
        return f"{self.nome} — R$ {self.preco:.2f} ({self.estoque} un.)"

from datetime import datetime

# dentro da classe Loja:
    def definir_desconto(self, qtd):
        if qtd >= 20:
            return 0.15
        elif qtd >= 10:
            return 0.10
        return 0

    def registrar_venda(self, nome, qtd):
        produto = self.buscar(nome)
        if produto is None:
            print("Produto não encontrado!")
            return
        if not produto.vender(qtd):
            print(f"Quantidade inválida! Disponível: {produto.estoque}")
            return
        desconto = self.definir_desconto(qtd)
        total = qtd * produto.preco * (1 - desconto)
        data = datetime.now().strftime("%d/%m/%Y %H:%M")
        self.vendas.append({"produto": nome, "qtd": qtd,
                            "total": total, "data": data})
        print(f"Venda registrada! Total: R$ {total:.2f}")

# dentro da classe Loja:
    def relatorio(self):
        if len(self.vendas) == 0:
            print("Nenhuma venda registrada.")
            return
        faturamento = 0
        unidades = 0
        por_produto = {}
        for venda in self.vendas:
            faturamento = faturamento + venda["total"]
            unidades = unidades + venda["qtd"]
            nome = venda["produto"]
            por_produto[nome] = por_produto.get(nome, 0) + venda["qtd"]

        print(f"Faturamento: R$ {faturamento:.2f}")
        print(f"Canecas vendidas: {unidades}")

        campeao = ""
        maior = 0
        for nome, qtd in por_produto.items():
            if qtd > maior:
                maior = qtd
                campeao = nome
        print(f"Campeão: {campeao} ({maior} un.)")

        print("Últimas vendas:")
        for venda in self.vendas:
            print(f"{venda['data']} — {venda['qtd']}x {venda['produto']} — R$ {venda['total']:.2f}")

import json
from datetime import datetime

class Produto:
    def __init__(self, nome, preco, estoque):
        self.nome = nome
        self.preco = preco
        self.estoque = estoque

    def vender(self, qtd):
        if qtd <= 0 or qtd > self.estoque:
            return False
        self.estoque = self.estoque - qtd
        return True

    def to_dict(self):
        return {"nome": self.nome, "preco": self.preco, "estoque": self.estoque}

    def __str__(self):
        return f"{self.nome} — R$ {self.preco:.2f} ({self.estoque} un.)"

class Loja:
    def __init__(self):
        self.produtos = []
        self.vendas = []

    def adicionar(self, produto):
        self.produtos.append(produto)

    def buscar(self, nome):
        for produto in self.produtos:
            if produto.nome == nome:
                return produto
        return None

    def listar(self):
        print("--- CATÁLOGO ---")
        for produto in self.produtos:
            print(produto)

    def definir_desconto(self, qtd):
        if qtd >= 20:
            return 0.15
        elif qtd >= 10:
            return 0.10
        return 0

    def registrar_venda(self, nome, qtd):
        produto = self.buscar(nome)
        if produto is None:
            print("Produto não encontrado!")
            return
        if not produto.vender(qtd):
            print(f"Quantidade inválida! Disponível: {produto.estoque}")
            return
        desconto = self.definir_desconto(qtd)
        total = qtd * produto.preco * (1 - desconto)
        data = datetime.now().strftime("%d/%m/%Y %H:%M")
        self.vendas.append({"produto": nome, "qtd": qtd,
                            "total": total, "data": data})
        print(f"Venda registrada! Total: R$ {total:.2f}")

    def relatorio(self):
        if len(self.vendas) == 0:
            print("Nenhuma venda registrada.")
            return
        faturamento = 0
        por_produto = {}
        for venda in self.vendas:
            faturamento = faturamento + venda["total"]
            por_produto[venda["produto"]] = por_produto.get(venda["produto"], 0) + venda["qtd"]
        print(f"Faturamento: R$ {faturamento:.2f}")
        campeao = ""
        maior = 0
        for nome, qtd in por_produto.items():
            if qtd > maior:
                maior = qtd
                campeao = nome
        print(f"Campeão: {campeao} ({maior} un.)")

    def salvar(self):
        dados = []
        for produto in self.produtos:
            dados.append(produto.to_dict())
        with open("loja.json", "w", encoding="utf-8") as arquivo:
            json.dump(dados, arquivo, indent=2, ensure_ascii=False)
        print("Loja salva ✔")

    def carregar(self):
        try:
            with open("loja.json", "r", encoding="utf-8") as arquivo:
                dados = json.load(arquivo)
            for d in dados:
                self.adicionar(Produto(d["nome"], d["preco"], d["estoque"]))
        except FileNotFoundError:
            self.adicionar(Produto("caneca branca", 25.0, 10))
            self.adicionar(Produto("caneca preta", 27.5, 8))
            self.adicionar(Produto("azulejo", 40.0, 5))

def pedir_inteiro(mensagem):
    while True:
        try:
            return int(input(mensagem))
        except ValueError:
            print("Digite um número inteiro!")

# ===== PROGRAMA PRINCIPAL =====
loja = Loja()
loja.carregar()

while True:
    print()
    print("=== TUA CANECA 2.0 ===")
    print("1-Catálogo  2-Venda  3-Relatório  4-Salvar e sair")
    opcao = pedir_inteiro("Opção: ")

    if opcao == 1:
        loja.listar()
    elif opcao == 2:
        nome = input("Produto: ")
        qtd = pedir_inteiro("Quantidade: ")
        loja.registrar_venda(nome, qtd)
    elif opcao == 3:
        loja.relatorio()
    elif opcao == 4:
        loja.salvar()
        print("Até logo, e parabéns pela formação! 🎓")
        break
    else:
        print("Opção inválida!")
