Só abra depois de tentar!

Os 5 bugs: (1) falta : após def calcular_media(notas) → SyntaxError. (2) for nota in nota deveria ser in notas. (3) soma =+ nota é o traiçoeiro: NÃO dá erro! Significa soma = +nota (atribui o valor positivo) — o certo é +=. (4) falta fechar o parêntese em len(notas. (5) Media com maiúscula não existe — Python diferencia maiúsculas.

solucao_ex02.py
def calcular_media(notas):
    soma = 0
    for nota in notas:
        soma += nota
    media = soma / len(notas)
    return media

print(calcular_media([8, 7, 9]))

Etapa 3: registrar venda 💰

⭐⭐⭐ desafio
🎯 Objetivo

O coração do sistema: validação, desconto progressivo e baixa no estoque.

🧭 Passo a passo
  1. Recupere definir_desconto(quantidade) da Semana 6 (20+ → 15% · 10+ → 10%)

  3. Valide: cor existe? quantidade entre 1 e o disponível? Se não → avise e return

💡 O return de saída rápida
Repare no padrão: validou, deu ruim, return — a função termina ali e o menu continua. Sem else aninhado em else. Código profissional adora essas "saídas rápidas".

etapa3.py
def definir_desconto(quantidade):
    if quantidade >= 20:
        return 0.15
    elif quantidade >= 10:
        return 0.10
    return 0

def registrar_venda(estoque, vendas):
    cor = input("Cor: ")
    if cor not in estoque:
        print("Cor não cadastrada!")
        return
    quantidade = int(input("Quantidade: "))
    if quantidade <= 0 or quantidade > estoque[cor]:
        print(f"Quantidade inválida! Disponível: {estoque[cor]}")
        return
    desconto = definir_desconto(quantidade)
    total = quantidade * PRECO * (1 - desconto)
    estoque[cor] = estoque[cor] - quantidade
    vendas.append({"cor": cor, "qtd": quantidade, "total": total})
    print(f"Venda registrada! Total: R$ {total:.2f}")

Etapa 4: o relatório 📊

⭐⭐⭐ desafio
🎯 Objetivo

Transformar a lista de vendas em informação: totais e a cor campeã.

🧭 Passo a passo
  1. Crie def relatorio(vendas): — se a lista estiver vazia, avise e return
  2. Percorra as vendas acumulando faturamento e total de canecas
  3. Use um dict contador (.get(cor, 0), Semana 5!) para somar canecas por cor
  4. Encontre a mais vendida com o padrão maior/menor da Semana 3
✅ Exemplo de execução
terminal
Faturamento: R$ 345.00
Canecas vendidas: 15
Mais vendida: preta (12 un.)
🔑 Ver solução

Só abra depois de tentar!

etapa4.py
def relatorio(vendas):
    if len(vendas) == 0:
        print("Nenhuma venda registrada.")
        return
    faturamento = 0
    canecas = 0
    por_cor = {}
    for venda in vendas:
        faturamento = faturamento + venda["total"]
        canecas = canecas + venda["qtd"]
        cor = venda["cor"]
        por_cor[cor] = por_cor.get(cor, 0) + venda["qtd"]

    print(f"Faturamento: R$ {faturamento:.2f}")
    print(f"Canecas vendidas: {canecas}")

    mais_vendida = ""
    maior = 0
    for cor, qtd in por_cor.items():
        if qtd > maior:
            maior = qtd
            mais_vendida = cor
    print(f"Mais vendida: {mais_vendida} ({maior} un.)")

Etapa 5: estoque que sobrevive 💾

🎯 Objetivo

Persistência: o estoque é carregado ao abrir e salvo ao sair.

etapa5.py
def salvar_estoque(estoque):
    with open("estoque.txt", "w", encoding="utf-8") as arquivo:
        for cor, qtd in estoque.items():
            arquivo.write(f"{cor};{qtd}\n")

def carregar_estoque():
    try:
        estoque = {}
        with open("estoque.txt", "r", encoding="utf-8") as arquivo:
            for linha in arquivo:
                cor, qtd = linha.strip().split(";")
                estoque[cor] = int(qtd)
        return estoque
    except FileNotFoundError:
        return {"branca": 10, "preta": 8, "azul": 5}

PRECO = 25.00

def pedir_inteiro(mensagem):
    while True:
        try:
            return int(input(mensagem))
        except ValueError:
            print("Digite um número inteiro!")

def carregar_estoque():
    try:
        estoque = {}
        with open("estoque.txt", "r", encoding="utf-8") as arquivo:
            for linha in arquivo:
                cor, qtd = linha.strip().split(";")
                estoque[cor] = int(qtd)
        return estoque
    except FileNotFoundError:
        return {"branca": 10, "preta": 8, "azul": 5}

def salvar_estoque(estoque):
    with open("estoque.txt", "w", encoding="utf-8") as arquivo:
        for cor, qtd in estoque.items():
            arquivo.write(f"{cor};{qtd}\n")

def consultar(estoque):
    print("--- ESTOQUE ---")
    for cor, qtd in estoque.items():
        print(f"{cor}: {qtd} unidades")

def definir_desconto(quantidade):
    if quantidade >= 20:
        return 0.15
    elif quantidade >= 10:
        return 0.10
    return 0

def registrar_venda(estoque, vendas):
    cor = input("Cor: ")
    if cor not in estoque:
        print("Cor não cadastrada!")
        return
    quantidade = pedir_inteiro("Quantidade: ")
    if quantidade <= 0 or quantidade > estoque[cor]:
        print(f"Quantidade inválida! Disponível: {estoque[cor]}")
        return
    desconto = definir_desconto(quantidade)
    total = quantidade * PRECO * (1 - desconto)
    estoque[cor] = estoque[cor] - quantidade
    vendas.append({"cor": cor, "qtd": quantidade, "total": total})
    print(f"Venda registrada! Total: R$ {total:.2f}")

def relatorio(vendas):
    if len(vendas) == 0:
        print("Nenhuma venda registrada.")
        return
    faturamento = 0
    canecas = 0
    por_cor = {}
    for venda in vendas:
        faturamento = faturamento + venda["total"]
        canecas = canecas + venda["qtd"]
        por_cor[venda["cor"]] = por_cor.get(venda["cor"], 0) + venda["qtd"]
    print(f"Faturamento: R$ {faturamento:.2f}")
    print(f"Canecas vendidas: {canecas}")
    mais_vendida = ""
    maior = 0
    for cor, qtd in por_cor.items():
        if qtd > maior:
            maior = qtd
            mais_vendida = cor

estoque = carregar_estoque()
vendas = []

while True:
    print()
    print("=== SISTEMA TUA CANECA ===")
    print("1-Estoque  2-Venda  3-Relatório  4-Salvar e sair")
