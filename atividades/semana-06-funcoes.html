Os 5 bugs: (1) faltou newline="" (e o encoding="utf-8") no open — linhas em branco no Windows. (2) writerow("nome", "total") passa DOIS argumentos: o writerow recebe UMA LISTA — writerow(["nome", "total"]). (3) os e shutil usados sem import! (4) os.makedirs("saida") sem exist_ok=True quebra na 2ª execução. (5) Caminho com "\\" na mão em vez de os.path.join — funciona só no Windows e é frágil.

solucao_ex02.py
import os
import csv
import shutil

with open("relatorio.csv", "w", newline="", encoding="utf-8") as arquivo:
    escritor = csv.writer(arquivo)
    escritor.writerow(["nome", "total"])
    escritor.writerow(["Ana", 75.0])

os.makedirs("saida", exist_ok=True)
shutil.move("relatorio.csv", os.path.join("saida", "relatorio.csv"))

import os

PASTAS = ["dados", "relatorios", "backup"]

def importar_vendas():
    vendas = []
    caminho = os.path.join("dados", "vendas.csv")
    with open(caminho, "r", encoding="utf-8") as arquivo:
        leitor = csv.DictReader(arquivo)
        for linha in leitor:
            linha["qtd"] = int(linha["qtd"])
            linha["total"] = float(linha["total"])
            vendas.append(linha)
    return vendas

etapa3.py
def analisar(vendas):
    faturamento = 0
    unidades = 0
    por_produto = {}
    for venda in vendas:
        faturamento = faturamento + venda["total"]
        unidades = unidades + venda["qtd"]
        nome = venda["produto"]
        por_produto[nome] = por_produto.get(nome, 0) + venda["qtd"]

    campeao = ""
    maior = 0
    for nome, qtd in por_produto.items():
        if qtd > maior:
            maior = qtd
            campeao = nome

    return {
        "faturamento": faturamento,
        "unidades": unidades,
        "campeao": campeao,
        "ticket": faturamento / len(vendas)
    }

etapa4.py
import json
from datetime import datetime

def cotacao_segura():
    try:
        import requests
        url = "https://economia.awesomeapi.com.br/json/last/USD-BRL"
        cotacao = float(requests.get(url, timeout=5).json()["USDBRL"]["bid"])
        cache = {"cotacao": cotacao,
                 "consultado_em": datetime.now().strftime("%d/%m/%Y %H:%M")}
        with open("cache_dolar.json", "w", encoding="utf-8") as arquivo:
            json.dump(cache, arquivo, indent=2, ensure_ascii=False)
        return cotacao
    except Exception:
        try:
            with open("cache_dolar.json", "r", encoding="utf-8") as arquivo:
                cache = json.load(arquivo)
            print(f"(offline) Usando cache de {cache['consultado_em']}")
            return cache["cotacao"]
        except FileNotFoundError:
            print("(offline, sem cache) Cotação padrão R$ 5.00")
            return 5.00

etapa5.py
from openpyxl import Workbook
from openpyxl.styles import Font

def gerar_relatorio(vendas, resumo, dolar):
    planilha = Workbook()
    aba = planilha.active
    aba.title = "Vendas"

    aba.append(["Data", "Produto", "Qtd", "Total (R$)"])
    for celula in aba[1]:
        celula.font = Font(bold=True)

    for venda in vendas:
        aba.append([venda["data"], venda["produto"],
                    venda["qtd"], venda["total"]])

    linha_total = len(vendas) + 2
    aba[f"C{linha_total}"] = "TOTAL:"
    aba[f"D{linha_total}"] = f"=SUM(D2:D{len(vendas) + 1})"

    aba.append([])
    aba.append(["Campeão", resumo["campeao"]])
    aba.append(["Ticket médio (R$)", round(resumo["ticket"], 2)])
    aba.append(["Faturamento (US$)", round(resumo["faturamento"] / dolar, 2)])

    nome = datetime.now().strftime("relatorio_%Y-%m-%d_%H-%M.xlsx")
    caminho = os.path.join("relatorios", nome)
    planilha.save(caminho)
    print(f"Relatório salvo: {caminho} ✔")

import os
import csv
import json
import shutil
from datetime import datetime
from openpyxl import Workbook
from openpyxl.styles import Font

PASTAS = ["dados", "relatorios", "backup"]

def preparar_pastas():
    for pasta in PASTAS:
        os.makedirs(pasta, exist_ok=True)

def gerar_exemplo():
    caminho = os.path.join("dados", "vendas.csv")
    if os.path.exists(caminho):
        return
    campos = ["data", "produto", "qtd", "total"]
    exemplos = [
        {"data": "10/06/2026", "produto": "caneca branca", "qtd": 3, "total": 75.0},
        {"data": "10/06/2026", "produto": "caneca preta", "qtd": 12, "total": 270.0},
        {"data": "11/06/2026", "produto": "azulejo", "qtd": 2, "total": 80.0},
        {"data": "12/06/2026", "produto": "caneca preta", "qtd": 5, "total": 137.5},
    ]
    with open(caminho, "w", newline="", encoding="utf-8") as arquivo:
        escritor = csv.DictWriter(arquivo, fieldnames=campos)
        escritor.writeheader()
        for linha in exemplos:
            escritor.writerow(linha)

def importar_vendas():
    vendas = []
    with open(os.path.join("dados", "vendas.csv"), "r", encoding="utf-8") as arquivo:
        leitor = csv.DictReader(arquivo)
        for linha in leitor:
            linha["qtd"] = int(linha["qtd"])
            linha["total"] = float(linha["total"])
            vendas.append(linha)
    return vendas

def analisar(vendas):
    faturamento = 0
    unidades = 0
    por_produto = {}
    for venda in vendas:
        faturamento = faturamento + venda["total"]
        unidades = unidades + venda["qtd"]
        por_produto[venda["produto"]] = por_produto.get(venda["produto"], 0) + venda["qtd"]
    campeao = ""
    maior = 0
    for nome, qtd in por_produto.items():
        if qtd > maior:
            maior = qtd
            campeao = nome
    return {"faturamento": faturamento, "unidades": unidades,
            "campeao": campeao, "ticket": faturamento / len(vendas)}

def cotacao_segura():
    try:
        import requests
        url = "https://economia.awesomeapi.com.br/json/last/USD-BRL"
        cotacao = float(requests.get(url, timeout=5).json()["USDBRL"]["bid"])
        cache = {"cotacao": cotacao,
                 "consultado_em": datetime.now().strftime("%d/%m/%Y %H:%M")}
        with open("cache_dolar.json", "w", encoding="utf-8") as arquivo:
            json.dump(cache, arquivo, indent=2, ensure_ascii=False)
        return cotacao
    except Exception:
        try:
            with open("cache_dolar.json", "r", encoding="utf-8") as arquivo:
                cache = json.load(arquivo)
            print(f"(offline) Usando cache de {cache['consultado_em']}")
            return cache["cotacao"]
        except FileNotFoundError:
            print("(offline, sem cache) Cotação padrão R$ 5.00")
            return 5.00

def gerar_relatorio(vendas, resumo, dolar):
    planilha = Workbook()
    aba = planilha.active
    aba.title = "Vendas"
    aba.append(["Data", "Produto", "Qtd", "Total (R$)"])
    for celula in aba[1]:
        celula.font = Font(bold=True)
    for venda in vendas:
        aba.append([venda["data"], venda["produto"], venda["qtd"], venda["total"]])
    linha_total = len(vendas) + 2
    aba[f"C{linha_total}"] = "TOTAL:"
    aba[f"D{linha_total}"] = f"=SUM(D2:D{len(vendas) + 1})"
    aba.append([])
    aba.append(["Campeão", resumo["campeao"]])
    aba.append(["Ticket médio (R$)", round(resumo["ticket"], 2)])
    aba.append(["Faturamento (US$)", round(resumo["faturamento"] / dolar, 2)])
    nome = datetime.now().strftime("relatorio_%Y-%m-%d_%H-%M.xlsx")
    caminho = os.path.join("relatorios", nome)
    planilha.save(caminho)
    print(f"Relatório salvo: {caminho} ✔")

def fazer_backup():
    nome = datetime.now().strftime("backup_%Y-%m-%d_%H-%M-%S")
    destino = os.path.join("backup", nome)
    shutil.copytree("dados", destino)
    print(f"Backup criado: {destino} ✔")
